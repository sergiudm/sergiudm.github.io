{{- /* A day (or any span) laid out on a time axis, one row per kind. {{< timeline start="07:00" end="24:00" title="A Tuesday" caption="..." size="wide" >}} 08:30-09:20 | potato | focus 13:20 | napping | reminder {{< /timeline >}} Each line: "HH:MM[-HH:MM] | label | kind". Ranges draw bars; single times draw markers. */ -}} {{- $start := partial "time-to-min.html" (.Get "start" | default "06:00") -}} {{- $end := partial "time-to-min.html" (.Get "end" | default "24:00") -}} {{- $span := sub $end $start -}} {{- $items := slice -}} {{- $kinds := slice -}} {{- range split (strings.TrimSpace .Inner) "\n" -}} {{- $line := strings.TrimSpace . -}} {{- if $line -}} {{- $f := split $line "|" -}} {{- $when := strings.TrimSpace (index $f 0) -}} {{- $label := "" -}} {{- if ge (len $f) 2 }}{{ $label = strings.TrimSpace (index $f 1) }}{{ end -}} {{- $kind := "event" -}} {{- if ge (len $f) 3 }}{{ $kind = strings.TrimSpace (index $f 2) }}{{ end -}} {{- $t := split $when "-" -}} {{- $s := partial "time-to-min.html" (index $t 0) -}} {{- $e := -1 -}} {{- if ge (len $t) 2 }}{{ $e = partial "time-to-min.html" (index $t 1) }}{{ end -}} {{- $items = $items | append (dict "s" $s "e" $e "label" $label "kind" $kind "when" (replace $when "-" "–")) -}} {{- if not (in $kinds $kind) }}{{ $kinds = $kinds | append $kind }}{{ end -}} {{- end -}} {{- end -}} {{- $step := cond (gt $span 720) 120 60 -}}
